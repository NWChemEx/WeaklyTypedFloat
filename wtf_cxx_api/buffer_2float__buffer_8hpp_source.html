601template<typename TupleType>
602auto make_float_buffer(std::vector<FloatBuffer::value_type> buffer) {
603 FloatBuffer rv;
604 if(buffer.size() == 0) return rv;
605 auto initializer = [&](auto value0) {
606 using T = std::decay_t<decltype(value0)>;
607 std::vector<T> temp_buffer(buffer.size());
608 rv = FloatBuffer(std::move(temp_buffer));
609 rv.at(0) = value0;
610 };
611 fp::visit_float<TupleType>(initializer, buffer[0]);
612 for(std::size_t i = 1; i < buffer.size(); ++i) {
613 auto appender = [&](auto value) { rv.at(i) = value; };
614 fp::visit_float<TupleType>(appender, buffer[i]);
615 }
616 return rv;
617}
