605template<typename TupleType>
606auto make_float_buffer(std::vector<FloatBuffer::value_type> buffer) {
607 FloatBuffer rv;
608 if(buffer.size() == 0) return rv;
609 auto initializer = [&](auto value0) {
610 using T = std::decay_t<decltype(value0)>;
611 std::vector<T> temp_buffer(buffer.size());
612 rv = FloatBuffer(std::move(temp_buffer));
613 rv.at(0) = value0;
614 };
615 fp::visit_float<TupleType>(initializer, buffer[0]);
616 for(std::size_t i = 1; i < buffer.size(); ++i) {
617 auto appender = [&](auto value) { rv.at(i) = value; };
618 fp::visit_float<TupleType>(appender, buffer[i]);
619 }
620 return rv;
621}

717template<typename TupleType = wtf::default_fp_types>
718FloatBuffer concatenate(std::span<const FloatBuffer> buffers) {
719 FloatBuffer::size_type n = 0;
720 const FloatBuffer* first = nullptr;
721 for(const auto& buffer : buffers) {
722 n += buffer.size();
723 if(first == nullptr && buffer.size() > 0) first = &buffer;
724 }
725 if(first == nullptr) return FloatBuffer{};
726
727 // The first non-empty buffer fixes the type of the result
728 auto concatenator = [&]<typename T>(std::span<T>) {
729 using clean_type = std::remove_const_t<T>;
730 std::vector<clean_type> rv;
731 rv.reserve(n);
732 auto appender = [&rv]<typename U>(std::span<U> values) {
733 if constexpr(std::is_same_v<std::remove_const_t<U>, clean_type>) {
734 rv.insert(rv.end(), values.begin(), values.end());
735 } else {
736 throw std::runtime_error(
737 "wtf::buffer::concatenate: buffers must all hold the same "
738 "type.");
739 }
740 };
741 for(const auto& buffer : buffers) {
742 if(buffer.size() == 0) continue;
743 visit_contiguous_buffer<TupleType>(appender, buffer);
744 }
745 return FloatBuffer(std::move(rv));
746 };
747 return visit_contiguous_buffer<TupleType>(concatenator, *first);
748}
