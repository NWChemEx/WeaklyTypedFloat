333 return wtf::cast::detail_::restore<detail_::ContiguousModel, T>(

487 void reserve(size_type n) {
488 using clean_t = std::decay_t<T>;
489 if(!is_holding_()) { *this = FloatBuffer(std::vector<clean_t>{}); }

602template<typename TupleType>
603auto make_float_buffer(std::vector<FloatBuffer::value_type> buffer) {
604 FloatBuffer rv;
605 if(buffer.size() == 0) return rv;
606 auto initializer = [&](auto value0) {
607 using T = std::decay_t<decltype(value0)>;
608 std::vector<T> temp_buffer(buffer.size());
609 rv = FloatBuffer(std::move(temp_buffer));
610 rv.at(0) = value0;
611 };
612 fp::visit_float<TupleType>(initializer, buffer[0]);
613 for(std::size_t i = 1; i < buffer.size(); ++i) {
614 auto appender = [&](auto value) { rv.at(i) = value; };
615 fp::visit_float<TupleType>(appender, buffer[i]);
616 }
617 return rv;
618}
