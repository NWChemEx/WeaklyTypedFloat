555template<typename TupleType>
556auto make_float_buffer(std::vector<FloatBuffer::value_type> buffer) {
557 FloatBuffer rv;
558 if(buffer.size() == 0) return rv;
559 auto initializer = [&](auto value0) {
560 using T = std::decay_t<decltype(value0)>;
561 std::vector<T> temp_buffer(buffer.size());
562 rv = FloatBuffer(std::move(temp_buffer));
563 rv.at(0) = value0;
564 };
565 fp::visit_float<TupleType>(initializer, buffer[0]);
566 for(std::size_t i = 1; i < buffer.size(); ++i) {
567 auto appender = [&](auto value) { rv.at(i) = value; };
568 fp::visit_float<TupleType>(appender, buffer[i]);
569 }
570 return rv;
571}
