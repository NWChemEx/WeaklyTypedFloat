412IGNORE_DANGLING_REFERENCE T float_cast(Float& f) {
413 auto* p = f.m_holder_.get();
414 using derived_type = detail_::FloatModel<std::decay_t<T>>;
415 auto pderived = dynamic_cast<derived_type*>(p);
416 if(pderived == nullptr) {
417 throw std::runtime_error("wtf::float_cast: bad cast");
418 }
419 return *pderived->data();
420}
