66template<template<concepts::FloatingPoint> typename ModelType,
67 typename TypeTuple, typename T>
68auto downcast_arg(T&& holder) {
69 using mutable_variant_t =
70 decltype(downcaster<ModelType, TypeTuple>(holder));
71 using const_variant_t =
72 decltype(const_downcaster<ModelType, TypeTuple>(holder));
73 using flat_variant_t = variant_cat_t<mutable_variant_t, const_variant_t>;
74
75 auto flatten = [](auto* p) -> flat_variant_t { return flat_variant_t{p}; };
76
77 // Because we must call the correct downcaster based on cv-qualifiers that
78 // are only knowable at runtime, and because those downcasters all have
79 // different return types (and thus signatures) we must explicitly branch.
80 if(holder.is_const()) {
81 auto v =
82 const_downcaster<ModelType, TypeTuple>(std::forward<T>(holder));
83 return std::visit(flatten, v);
84 } else {
85 auto v = downcaster<ModelType, TypeTuple>(std::forward<T>(holder));
86 return std::visit(flatten, v);
87 }
88}

118template<template<concepts::FloatingPoint> typename ModelType,
119 typename TypeTuple, typename FxnType, typename... HolderTypes>
120auto dispatch(FxnType&& fxn, HolderTypes&&... holders) {
121 if constexpr(sizeof...(HolderTypes) == 0) {
122 return fxn();
123 } else {
124 // Braced-init-list guarantees left-to-right evaluation, preserving
125 // the old recursion's deterministic "leftmost failing holder
126 // throws" behavior (a bare pack-expanded call-argument list does
127 // not guarantee this).
128 std::tuple<decltype(downcast_arg<ModelType, TypeTuple>(
129 std::declval<HolderTypes&&>()))...>
130 variants{downcast_arg<ModelType, TypeTuple>(
131 std::forward<HolderTypes>(holders))...};
132
133 auto deref_and_call = [&fxn](auto*... ps) { return fxn(*ps...); };
134
135 return std::apply(
136 [&](auto&&... vs) { return std::visit(deref_and_call, vs...); },
137 std::move(variants));
138 }
139}
