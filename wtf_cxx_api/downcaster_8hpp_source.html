44template<typename VariantType, typename T, std::size_t... I>
45VariantType downcast_impl_(T&& holder, std::index_sequence<I...>) {
46 // Is holder a read-only object (regardless of whether it is holding a
47 // read-only type)?
48 using clean_type = std::remove_reference_t<T>;
49 static constexpr bool is_const = std::is_const_v<clean_type>;
50
51 std::optional<VariantType> result;
52
53 auto try_alternative =
54 [&]<std::size_t Idx>(std::integral_constant<std::size_t, Idx>) {
55 if(result.has_value()) return;
56
57 // This iteration we try to cast to TargetType
58 using TargetType = std::variant_alternative_t<Idx, VariantType>;
59
60 // Is TargetType a read-only type?
61 using clean_target_type = std::remove_pointer_t<TargetType>;
62 constexpr bool const_target = std::is_const_v<clean_target_type>;
63
64 if constexpr(is_const && !const_target) {
65 // Can't cast const to non-const
66 } else if(auto* p = dynamic_cast<TargetType>(&holder)) {
67 result = VariantType{p};
68 }
69 };
70
71 (try_alternative(std::integral_constant<std::size_t, I>{}), ...);
72
73 if(!result.has_value()) throw std::runtime_error("Type not in variant");
74 return std::move(*result);
75}

88template<typename VariantType, typename T>
89VariantType downcast_impl(T&& holder) {
90 constexpr auto n = std::variant_size_v<VariantType>;
91 return downcast_impl_<VariantType>(std::forward<T>(holder),
92 std::make_index_sequence<n>{});
93}

114template<template<concepts::FloatingPoint> typename ModelType, typename Types,
115 typename T>
116auto downcaster(T&& holder) {
117 constexpr bool is_const = std::is_const_v<std::remove_reference_t<T>>;
118 using variant_t = typename VariantFromTuple<ModelType, Types>::value;
119 using cvariant_t = typename ConstVariantFromTuple<ModelType, Types>::value;
120 using variant_type = std::conditional_t<is_const, cvariant_t, variant_t>;
121 return downcast_impl<variant_type>(std::forward<T>(holder));
122}
