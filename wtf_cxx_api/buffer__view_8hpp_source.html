283 bool operator==(const BufferView& other) const {

353 template<typename T>
354 std::span<T> value() {
355 if(!is_holding_()) return {};
356 return wtf::cast::detail_::restore<contiguous_model_type, T>(
357 holder_(),
358 "BufferView does not hold the requested floating-point type")
359 .span();
360 }
361
377 template<typename T>
378 std::span<const T> value() const {
379 if(!is_holding_()) return {};
380 // The underlying holder is never actually const (only the
381 // unique_ptr member is, and only because *this is const); casting
382 // away *this's constness lets us reuse the same restore() call the
383 // non-const overload uses, with the correct constness of T already
384 // baked into contiguous_model_type<T> via apply_const_t.
385 return wtf::cast::detail_::restore<contiguous_model_type, T>(
386 const_cast<BufferView*>(this)->holder_(),
387 "BufferView does not hold the requested floating-point type")
388 .span();
389 }

509template<typename T, typename Policy = fp::policies::Match,
510 typename TupleType = wtf::default_fp_types>
511std::vector<T> convert_to(BufferView<const fp::Float> buffer) {
512 std::vector<T> result;
513 result.reserve(buffer.size());
514 for(std::size_t i = 0; i < buffer.size(); ++i) {
515 result.push_back(fp::convert_to<T, Policy, TupleType>(buffer.at(i)));
516 }
517 return result;
518}
